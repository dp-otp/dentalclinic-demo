    <!-- Nanotech Hex Grid Animation -->
    <script type="module">
      import * as THREE from 'https://cdn.jsdelivr.net/npm/three@0.152.2/build/three.module.js';

      // Get canvas
      const canvas = document.getElementById('nanotech-canvas');
      if (!canvas) {
        console.error('Nanotech: Canvas not found');
      }

      // Scene setup
      const scene = new THREE.Scene();
      scene.background = null; // Transparent

      const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
      camera.position.z = 15;

      const renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: true, alpha: true });
      renderer.setSize(window.innerWidth, window.innerHeight);

      // Enable tone mapping for enhanced glow
      renderer.toneMapping = THREE.ACESFilmicToneMapping;
      renderer.toneMappingExposure = 1.5;

      // Hex grid parameters - well-distributed layout
      const hexRadius = 0.7;
      const spacing = 2.2;
      const stripWidth = 45;
      const stripHeight = 8;

      const hexagons = [];
      const particles = [];
      const lines = [];

      // Colors
      const colors = [
        new THREE.Color(0x00ffff), // Cyan
        new THREE.Color(0x00d4d4), // Turquoise
        new THREE.Color(0x00aaaa)  // Teal
      ];

      // Store hex positions for connecting lines
      const hexPositions = [];
      const minDistance = hexRadius * 2.5; // Minimum distance to prevent overlap

      // Create hexagons in diagonal strip pattern with collision detection
      const hexCount = 120;
      let attempts = 0;
      const maxAttempts = hexCount * 3;

      while (hexPositions.length < hexCount && attempts < maxAttempts) {
        attempts++;

        // Balanced culling
        if (Math.random() > 0.6) continue;

        // Position along diagonal strip (0 = bottom-left, 1 = top-right)
        const stripProgress = (attempts / maxAttempts);

        // Base diagonal position
        const diagonalBase = stripProgress * stripWidth - stripWidth / 2;

        // Random offset perpendicular to diagonal for strip thickness
        const perpOffset = (Math.random() - 0.5) * stripHeight;

        // Convert diagonal coordinates to x,y
        // Diagonal goes from bottom-left to top-right at 45 degrees
        const x = diagonalBase * 0.707 - perpOffset * 0.707;
        const y = diagonalBase * 0.707 + perpOffset * 0.707;

        // Check for collision with existing hexagons
        let tooClose = false;
        for (let j = 0; j < hexPositions.length; j++) {
          const dx = x - hexPositions[j].x;
          const dy = y - hexPositions[j].y;
          const dist = Math.sqrt(dx * dx + dy * dy);
          if (dist < minDistance) {
            tooClose = true;
            break;
          }
        }

        if (tooClose) continue;

        // Random depth
        const z = -Math.floor(Math.random() * 3) * 0.5;

        // Diagonal position for wave timing (normalized 0-1)
        const diagonalPos = stripProgress;

        // Store position immediately (for collision detection)
        const depthIndex = Math.floor(-z / 0.5);
        hexPositions.push({ x, y, z, diagonalPos, depthIndex });

        // Create complete hexagon (closed loop)
        const hexPoints = [];
        for (let j = 0; j <= 6; j++) {
          const angle = (Math.PI / 3) * (j % 6);
          const px = hexRadius * Math.cos(angle);
          const py = hexRadius * Math.sin(angle);
          hexPoints.push(new THREE.Vector3(px, py, 0));
        }

        const geometry = new THREE.BufferGeometry().setFromPoints(hexPoints);
        const color = colors[depthIndex];

        const material = new THREE.LineBasicMaterial({
          color: color,
          transparent: true,
          opacity: 0,
          blending: THREE.AdditiveBlending
        });

        const hex = new THREE.Line(geometry, material);
        hex.position.set(x, y, z);
        scene.add(hex);

        // Create subtle particle at center
        const particleGeometry = new THREE.BufferGeometry();
        particleGeometry.setAttribute('position', new THREE.BufferAttribute(new Float32Array([0, 0, 0]), 3));

        const particleSize = 0.08 + Math.random() * 0.04;
        const particleMaterial = new THREE.PointsMaterial({
          color: color.clone().multiplyScalar(1.5),
          size: particleSize,
          transparent: true,
          opacity: 0,
          blending: THREE.AdditiveBlending,
          sizeAttenuation: true
        });

        const particle = new THREE.Points(particleGeometry, particleMaterial);
        particle.position.set(x, y, z);
        scene.add(particle);

        hexagons.push({
          mesh: hex,
          baseOpacity: 0.6,
          diagonalPos: diagonalPos
        });

        particles.push({
          mesh: particle,
          baseOpacity: 0.7,
          diagonalPos: diagonalPos
        });
      }

      // Create connecting lines
      for (let i = 0; i < hexPositions.length; i++) {
        const pos1 = hexPositions[i];

        for (let j = i + 1; j < hexPositions.length; j++) {
          const pos2 = hexPositions[j];

          const dist = Math.sqrt(
            Math.pow(pos2.x - pos1.x, 2) +
            Math.pow(pos2.y - pos1.y, 2)
          );

          const depthDiff = Math.abs(pos2.z - pos1.z);

          // Fewer connecting lines
          if (dist < spacing * 1.1 && depthDiff <= 0.5 && Math.random() > 0.65) {
            const linePoints = [
              new THREE.Vector3(pos1.x, pos1.y, pos1.z),
              new THREE.Vector3(pos2.x, pos2.y, pos2.z)
            ];

            const lineGeometry = new THREE.BufferGeometry().setFromPoints(linePoints);
            const lineColor = colors[pos1.depthIndex];

            const lineOpacity = 0.15 + Math.random() * 0.15;
            const lineMaterial = new THREE.LineBasicMaterial({
              color: lineColor,
              transparent: true,
              opacity: lineOpacity,
              blending: THREE.AdditiveBlending
            });

            const line = new THREE.Line(lineGeometry, lineMaterial);
            scene.add(line);

            const avgDiagonal = (pos1.diagonalPos + pos2.diagonalPos) / 2;

            lines.push({
              mesh: line,
              baseOpacity: lineOpacity,
              diagonalPos: avgDiagonal
            });
          }
        }
      }

      console.log(`Created ${hexagons.length} hexagons, ${particles.length} particles, ${lines.length} lines`);

      // Simple animation with smooth fade in and scroll-based fade out
      let introProgress = 0;
      const introDuration = 2.0;
      let scrollProgress = 0;

      // Track scroll for fade out effect
      function updateScroll() {
        const maxScroll = window.innerHeight * 0.5;
        scrollProgress = Math.min(1, window.scrollY / maxScroll);
      }

      window.addEventListener('scroll', updateScroll, { passive: true });
      updateScroll();

      function animate(time) {
        requestAnimationFrame(animate);
        const seconds = time * 0.001;

        if (introProgress < 1.0) {
          introProgress = Math.min(1.0, introProgress + 0.016 / introDuration);
        }

        const deconstructEase = 1 - scrollProgress * scrollProgress;

        // Animate hexagons - simple smooth fade in with gentle breathing
        hexagons.forEach(({ mesh, baseOpacity, diagonalPos }) => {
          const waveDelay = diagonalPos * 0.3;
          const fadeDuration = 1.0;
          const localIntro = Math.max(0, Math.min(1, (introProgress - waveDelay) / fadeDuration));

          const fadeEase = localIntro * localIntro * localIntro * (localIntro * (localIntro * 6 - 15) + 10);

          // Gentle breathing animation
          const breathe = Math.sin(seconds * 0.2 + diagonalPos * 1.5) * 0.015 + 1.0;
          mesh.scale.set(breathe, breathe, 1);

          mesh.material.opacity = baseOpacity * fadeEase * deconstructEase;
        });

        // Animate particles - subtle fade in and gentle pulsing
        particles.forEach(({ mesh, baseOpacity, diagonalPos }) => {
          const waveDelay = diagonalPos * 0.3 + 0.15;
          const fadeDuration = 0.8;
          const localIntro = Math.max(0, Math.min(1, (introProgress - waveDelay) / fadeDuration));

          const fadeEase = localIntro * localIntro * localIntro * (localIntro * (localIntro * 6 - 15) + 10);

          // Subtle pulsing
          const pulse = Math.sin(seconds * 0.25 + diagonalPos * 2) * 0.08 + 1.0;
          mesh.scale.set(pulse, pulse, pulse);

          mesh.material.opacity = baseOpacity * fadeEase * deconstructEase;
        });

        // Animate connecting lines - simple fade in
        lines.forEach(({ mesh, baseOpacity, diagonalPos }) => {
          const waveDelay = diagonalPos * 0.3 + 0.2;
          const fadeDuration = 0.9;
          const localIntro = Math.max(0, Math.min(1, (introProgress - waveDelay) / fadeDuration));

          const fadeEase = localIntro * localIntro * localIntro * (localIntro * (localIntro * 6 - 15) + 10);

          // Subtle flow animation
          const flow = Math.sin(seconds * 0.18 + diagonalPos * 1.8) * 0.08 + 0.92;

          mesh.material.opacity = baseOpacity * fadeEase * flow * deconstructEase;
        });

        renderer.render(scene, camera);
      }

      animate(0);

      window.addEventListener('resize', () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
      });
    </script>

            hamburger.classList.toggle('active');
            navLinks.classList.toggle('active');
        }

        };

            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    entry.target.classList.add('visible');
